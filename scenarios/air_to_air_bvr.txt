# 3) 蓝方战斗机平台类型
################################################################################

platform_type BLUE_FIGHTER WSF_PLATFORM
   icon F-22

   radar_signature   FIGHTER_RADAR_SIG
   infrared_signature FIGHTER_IR_SIG
   optical_signature FIGHTER_OPTICAL_SIG

   mover WSF_AIR_MOVER
      maximum_speed 600 m/sec
      minimum_speed 120 m/sec
      default_radial_acceleration 9.0 g
   end_mover

   # 火控雷达,探测结果送入航迹处理器
   sensor radar FIRE_CONTROL_RADAR
      processor track-proc
   end_sensor

   # 航迹处理器:融合雷达探测为本地航迹
   processor track-proc WSF_TRACK_PROCESSOR
      update_interval 1.0 sec
   end_processor

   # 攻击处理器:进入包线即发射,带冷却避免重复发射
   processor attack-proc WSF_SCRIPT_PROCESSOR
      update_interval 1.0 sec

      script_variables
         double lastFireTime  = -100.0;
         double FIRE_COOLDOWN = 8.0;       # 两次发射最小间隔 (sec)
         double MAX_RANGE     = 55000.0;   # 最大发射距离 (m)
         double MIN_RANGE     = 3000.0;    # 最小发射距离 (m)
      end_script_variables

      on_update
         WsfWeapon weapon = PLATFORM.Weapon("aam");
         WsfLocalTrackList trackList = PLATFORM.MasterTrackList();

         if ((weapon.QuantityRemaining() > 0) && (trackList.TrackCount() > 0))
         {
            if (TIME_NOW - lastFireTime >= FIRE_COOLDOWN)
            {
               WsfTrack track = trackList.TrackEntry(0);
               double rng = PLATFORM.SlantRangeTo(track);
               if ((rng > MIN_RANGE) && (rng < MAX_RANGE))
               {
                  bool fired = weapon.Fire(track);
                  if (fired)
                  {
                     lastFireTime = TIME_NOW;
                     print(TIME_NOW, " [", PLATFORM.Name(), "] 发射 AAM -> ",
                           track.TargetName(), " 距离 ", rng, " m, 余弹 ",
                           weapon.QuantityRemaining());
                  }
               }
            }
         }
      end_on_update
   end_processor

   # 武器挂载
   weapon aam AAM
      quantity 4
      firing_interval 2.0 sec
   end_weapon
end_platform_type
