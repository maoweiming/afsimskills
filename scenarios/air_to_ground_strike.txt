# 空对地精确打击仿真 (Air-to-Ground Precision Strike / SEAD)

# 场景:1 架打击机摧毁 1 个地面 SAM / 雷达站
#   - blue-striker (蓝方):机载下视雷达探测地面目标 + GPS/激光制导炸弹 (类 GBU)
#   - red-sam-site (红方):地面防空 / 雷达站,打击目标

# 交战链路:打击机下视雷达探测地面目标 -> 建立航迹 -> 进入投放包线 -> 投放制导炸弹 -> 命中摧毁
#
# 运行: python scripts/run_mission.py scenarios/air_to_ground_strike.txt -es
################################################################################

   transmitter

# 2) GPS/激光制导炸弹 (类 GBU,无动力滑翔)

   velocity_pursuit_gain 5.0

################################################################################
# 3) 蓝方打击机平台类型
################################################################################

platform_type BLUE_STRIKER WSF_PLATFORM
   icon F-15E

   radar_signature GROUND_TARGET_SIG

   mover WSF_AIR_MOVER
      maximum_speed 500 m/sec
      minimum_speed 120 m/sec
      default_radial_acceleration 7.0 g
   end_mover

   # 下视打击雷达
   sensor radar STRIKE_RADAR
      processor track-proc
   end_sensor

   processor track-proc WSF_TRACK_PROCESSOR
      update_interval 1.0 sec
   end_processor

   # 投弹处理器:进入投放包线即投放,带冷却
   processor attack-proc WSF_SCRIPT_PROCESSOR
      update_interval 1.0 sec

      script_variables
         double lastReleaseTime = -100.0;
         double RELEASE_COOLDOWN = 5.0;
         double MAX_RANGE = 15000.0;   # 最大投放斜距 (m) — 滑翔弹射程
         double MIN_RANGE = 1000.0;    # 最小投放斜距 (m)
      end_script_variables

      on_update
         WsfWeapon weapon = PLATFORM.Weapon("gbu");
         WsfLocalTrackList trackList = PLATFORM.MasterTrackList();

         if ((weapon.QuantityRemaining() > 0) && (trackList.TrackCount() > 0))
         {
            if (TIME_NOW - lastReleaseTime >= RELEASE_COOLDOWN)
            {
               WsfTrack track = trackList.TrackEntry(0);
               double rng = PLATFORM.SlantRangeTo(track);
               if ((rng > MIN_RANGE) && (rng < MAX_RANGE))
               {
                  bool fired = weapon.Fire(track);
                  if (fired)
                  {
                     lastReleaseTime = TIME_NOW;
                     print(TIME_NOW, " [", PLATFORM.Name(), "] 投放 GBU -> ",
                           track.TargetName(), " 斜距 ", rng, " m, 余弹 ",
                           weapon.QuantityRemaining());
                  }
               }
            }
         }
      end_on_update
   end_processor

   weapon gbu GBU
      quantity 2
      firing_interval 3.0 sec
   end_weapon
end_platform_type
